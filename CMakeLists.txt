cmake_minimum_required(VERSION 3.20)
if(APPLE)
  set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "")
  set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "")
endif()
project(bazz CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)

add_library(bazz_graph STATIC src/graph.cpp)
target_include_directories(bazz_graph PUBLIC src)
set_target_properties(bazz_graph PROPERTIES POSITION_INDEPENDENT_CODE ON)

if(WIN32)
  set(dev_default OFF)  # the hot-reload loader is POSIX-only
else()
  set(dev_default ON)
endif()
option(BAZZ_DEV "Hot-reloadable core and tools/reaper file bridge (off for release builds)" ${dev_default})
if(APPLE)
  set(PLUGIN_SUFFIX ".dylib")
elseif(WIN32)
  set(PLUGIN_SUFFIX ".dll")
  add_compile_definitions(NOMINMAX _CRT_SECURE_NO_WARNINGS)
else()
  set(PLUGIN_SUFFIX ".so")
  add_compile_definitions(SWELL_PROVIDED_BY_APP WDL_NO_DEFINE_MINMAX)
endif()
if(MSVC)
  add_compile_options(/utf-8 /W3)
else()
  add_compile_options(-Wall -Wextra)
endif()

# Fonts compiled into the binary.
set(FONTS_H "${CMAKE_BINARY_DIR}/generated/fonts.h")
file(WRITE "${FONTS_H}" "#pragma once\n")
foreach(pair "kFontUi:WineTahoma.ttf" "kFontBold:WineTahomaBold.ttf")
  string(REPLACE ":" ";" pair "${pair}")
  list(GET pair 0 sym)
  list(GET pair 1 file)
  file(READ "${CMAKE_SOURCE_DIR}/vendor/fonts/${file}" hex HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," bytes "${hex}")
  file(APPEND "${FONTS_H}" "inline const unsigned char ${sym}[] = {${bytes}};\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/vendor/fonts/${file}")
endforeach()

set(CORE_SOURCES src/core.cpp src/reaper_graph.cpp src/ui.cpp src/gfx.cpp)
set(PLUGINS reaper_bazz)
if(BAZZ_DEV)
  list(APPEND PLUGINS bazz_core)
endif()

foreach(mod ${PLUGINS})
  add_library(${mod} MODULE)
  target_include_directories(${mod} PRIVATE src)
  target_include_directories(${mod} SYSTEM PRIVATE vendor/reaper-sdk/sdk vendor/reaper-sdk/WDL)
  set_target_properties(${mod} PROPERTIES PREFIX "" SUFFIX "${PLUGIN_SUFFIX}")
endforeach()

if(BAZZ_DEV)
  set(CORE bazz_core)
  target_sources(reaper_bazz PRIVATE src/loader.cpp)
  target_compile_definitions(reaper_bazz PRIVATE BAZZ_DEV
    BAZZ_CORE_PATH="$<TARGET_FILE:bazz_core>"
    BAZZ_BRIDGE_DIR="${CMAKE_SOURCE_DIR}/.bridge")
  add_dependencies(reaper_bazz bazz_core)
  target_compile_definitions(bazz_core PRIVATE BAZZ_DEV)
else()
  set(CORE reaper_bazz)
  target_sources(reaper_bazz PRIVATE src/loader.cpp)
endif()
target_sources(${CORE} PRIVATE ${CORE_SOURCES})
target_include_directories(${CORE} PRIVATE "${CMAKE_BINARY_DIR}/generated")
target_include_directories(${CORE} SYSTEM PRIVATE vendor/reaimgui)
target_link_libraries(${CORE} PRIVATE bazz_graph)

enable_testing()
add_executable(graph_tests tests/graph_tests.cpp)
target_link_libraries(graph_tests PRIVATE bazz_graph)
target_include_directories(graph_tests PRIVATE vendor/doctest)
if(NOT MSVC)
  target_compile_options(graph_tests PRIVATE -Wno-missing-field-initializers)  # tests build machines with brace lists
endif()
add_test(NAME graph_tests COMMAND graph_tests)

if(APPLE)
  set(REAPER_USERPLUGINS "$ENV{HOME}/Library/Application Support/REAPER/UserPlugins")
elseif(WIN32)
  set(REAPER_USERPLUGINS "$ENV{APPDATA}/REAPER/UserPlugins")
else()
  set(REAPER_USERPLUGINS "$ENV{HOME}/.config/REAPER/UserPlugins")
endif()
add_custom_target(install_link
  COMMAND ${CMAKE_COMMAND} -E create_symlink $<TARGET_FILE:reaper_bazz> "${REAPER_USERPLUGINS}/reaper_bazz${PLUGIN_SUFFIX}"
  DEPENDS reaper_bazz)
